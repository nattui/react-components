import { IconCoinsOutline18, IconPlusOutline18, IconRadioOutline18 } from "@nattstack/icons"
import {
  Button,
  Column,
  Label,
  Menu,
  MenuContent,
  MenuItem,
  MenuTrigger,
  Spacer,
} from "@nattstack/ui"

# Menu

Menus display a list of actions or choices anchored to a trigger.

```jsx
// component

<Column>
  <Label>Dropdown menu</Label>
  <Spacer height={4} />
  <Menu>
    <MenuTrigger
      render={
        <Button iconStart={<IconPlusOutline18 />} size={36} variant="secondary">
          Create
        </Button>
      }
    />
    <MenuContent align="start" className="w-144">
      <MenuItem onClick={() => console.log("New coin")}>
        <IconCoinsOutline18 />
        {"New coin"}
      </MenuItem>
      <MenuItem onClick={() => console.log("Go live")}>
        <IconRadioOutline18 />
        {"Go live"}
      </MenuItem>
    </MenuContent>
  </Menu>
</Column>
```
